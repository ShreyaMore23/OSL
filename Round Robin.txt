Round Robin
#include <iostream>
using namespace std;

int main() {
    int n, tq;
    cin >> n >> tq;

    int at[n], bt[n], rt[n], ct[n], tat[n], wt[n];

    for(int i = 0; i < n; i++) {
        cin >> at[i] >> bt[i];
        rt[i] = bt[i];
    }

    int time = 0, completed = 0;

    while(completed < n) {
        for(int i = 0; i < n; i++) {
            if(rt[i] > 0 && at[i] <= time) {
                if(rt[i] > tq) {
                    time += tq;
                    rt[i] -= tq;
                } else {
                    time += rt[i];
                    ct[i] = time;
                    tat[i] = ct[i] - at[i];
                    wt[i] = tat[i] - bt[i];
                    rt[i] = 0;
                    completed++;
                }
            }
        }
    }

    return 0;
}
