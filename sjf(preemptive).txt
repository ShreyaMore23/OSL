#include <iostream>
using namespace std;

int main() {
    int n;
    cout << "Enter number of processes: ";
    cin >> n;

    int at[n], bt[n], rt[n], ct[n], tat[n], wt[n];

    for(int i = 0; i < n; i++) {
        cout << "Enter AT and BT for P" << i+1 << ": ";
        cin >> at[i] >> bt[i];
        rt[i] = bt[i];
    }

    int time = 0, complete = 0;

    while(complete != n) {
        int minRT = 9999, index = -1;

        for(int i = 0; i < n; i++) {
            if(at[i] <= time && rt[i] > 0 && rt[i] < minRT) {
                minRT = rt[i];
                index = i;
            }
        }

        if(index != -1) {
            rt[index]--;
            time++;

            if(rt[index] == 0) {
                complete++;
                ct[index] = time;
                tat[index] = ct[index] - at[index];
                wt[index] = tat[index] - bt[index];
            }
        } else {
            time++;
        }
    }

    cout << "\nPID AT BT CT TAT WT\n";
    for(int i = 0; i < n; i++) {
        cout << i+1 << " "
             << at[i] << " "
             << bt[i] << " "
             << ct[i] << " "
             << tat[i] << " "
             << wt[i] << endl;
    }

    return 0;
}
