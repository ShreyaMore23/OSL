Experiment 7 – Page Replacement (FIFO, LRU, Optimal)
#include <iostream>
#include <vector>
#include <queue>
#include <unordered_set>
using namespace std;

// FIFO
void FIFO(int pages[], int n, int capacity) {
    queue<int> q;
    unordered_set<int> s;
    int faults = 0;

    for(int i = 0; i < n; i++) {
        if(s.find(pages[i]) == s.end()) {
            if(s.size() == capacity) {
                int val = q.front();
                q.pop();
                s.erase(val);
            }
            q.push(pages[i]);
            s.insert(pages[i]);
            faults++;
        }
    }

    cout << "FIFO Page Faults = " << faults << endl;
}

// LRU
void LRU(int pages[], int n, int capacity) {
    vector<int> s;
    int faults = 0;

    for(int i = 0; i < n; i++) {
        int found = -1;

        for(int j = 0; j < s.size(); j++) {
            if(s[j] == pages[i]) {
                found = j;
                break;
            }
        }

        if(found != -1)
            s.erase(s.begin() + found);
        else {
            if(s.size() == capacity)
                s.erase(s.begin());
            faults++;
        }

        s.push_back(pages[i]);
    }

    cout << "LRU Page Faults = " << faults << endl;
}
// ---------------- OPTIMAL USING VECTOR ----------------

bool isPresent(vector<int> &frame, int page)
{
    for(int x : frame)
        if(x == page)
            return true;

    return false;
}

void Optimal(int pages[], int n, int capacity)
{
    vector<int> frame;
    int faults = 0;

    for(int i = 0; i < n; i++)
    {
        // If page not present
        if(!isPresent(frame, pages[i]))
        {
            // If space available
            if(frame.size() < capacity)
            {
                frame.push_back(pages[i]);
            }
            else
            {
                int index = -1, farthest = i;

                // Find page used farthest in future
                for(int j = 0; j < frame.size(); j++)
                {
                    int k;

                    for(k = i + 1; k < n; k++)
                    {
                        if(frame[j] == pages[k])
                            break;
                    }

                    if(k > farthest)
                    {
                        farthest = k;
                        index = j;
                    }
                }

                if(index == -1)
                    index = 0;

                frame[index] = pages[i]; // replace page
            }

            faults++;
        }
    }

    cout << "Optimal Page Faults = " << faults << endl;
}

// ---------------- MAIN FUNCTION ----------------

int main()
{
    int pages[] = {1, 3, 0, 3, 5, 6, 3};

    int n = sizeof(pages) / sizeof(pages[0]);

    int frames = 3;

    FIFO(pages, n, frames);

    LRU(pages, n, frames);

    Optimal(pages, n, frames);

    return 0;
}
