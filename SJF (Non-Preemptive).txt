SJF (Non-Preemptive)
#include <iostream>
using namespace std;

int main() {
    int n;
    cin >> n;

    int at[n], bt[n], ct[n], tat[n], wt[n];
    bool completed[n] = {false};

    for(int i = 0; i < n; i++)
        cin >> at[i] >> bt[i];

    int time = 0, done = 0;

    while(done < n) {
        int idx = -1, minBT = 9999;

        for(int i = 0; i < n; i++) {
            if(at[i] <= time && !completed[i] && bt[i] < minBT) {
                minBT = bt[i];
                idx = i;
            }
        }

        if(idx != -1) {
            time += bt[idx];
            ct[idx] = time;
            tat[idx] = ct[idx] - at[idx];
            wt[idx] = tat[idx] - bt[idx];
            completed[idx] = true;
            done++;
        } else {
            time++;
        }
    }

    return 0;
}
